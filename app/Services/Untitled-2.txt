<?php

namespace App\Services;

use App\Repositories\ResidentRepositoryInterface;

class ResidentSearchService
{
    private ResidentRepositoryInterface $repository;

    public function __construct(ResidentRepositoryInterface $repository)
    {
        $this->repository = $repository;
    }

    public function listResidents(): array
    {
        return $this->repository->findAll();
    }

    public function searchResidents(?string $searchTerm): array
    {
        $trimmedTerm = trim((string) $searchTerm);

        if ($trimmedTerm === '') {
            return $this->listResidents();
        }

        return $this->repository->searchByName($trimmedTerm);
    }
}
